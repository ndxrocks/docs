```python
"""
In this example, we upload a PDF file to Anthropic directly and then use it as an input to an agent.
"""

from pathlib import Path

from kern.agent import Agent
from kern.media import Image
from kern.models.anthropic import Claude
from kern.utils.media import download_file
from anthropic import Anthropic

# ---------------------------------------------------------------------------
# Create Agent
# ---------------------------------------------------------------------------

img_path = Path(__file__).parent.joinpath("kern-intro.png")

# Download the file using the download_file function
download_file(
    "https://kern-public.s3.us-east-1.amazonaws.com/images/kern-intro.png",
    str(img_path),
)

# Initialize Anthropic client
client = Anthropic()

agent = Agent(
    model=Claude(id="claude-sonnet-4-20250514"),
    markdown=True,
)

agent.print_response(
    "What does the attached image say.",
    images=[Image(filepath=img_path)],
)

# ---------------------------------------------------------------------------
# Run Agent
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    pass
```
