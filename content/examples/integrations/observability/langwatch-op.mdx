```python
"""
LangWatch Integration
=====================

Demonstrates instrumenting an Kern agent and sending traces to LangWatch.
"""

import langwatch
from kern.agent import Agent
from kern.models.openai import OpenAIChat
from kern.tools.yfinance import YFinanceTools
from openinference.instrumentation.kern import AgnoInstrumentor

# ---------------------------------------------------------------------------
# Setup
# ---------------------------------------------------------------------------
# Initialize LangWatch and instrument Kern
langwatch.setup(instrumentors=[AgnoInstrumentor()])


# ---------------------------------------------------------------------------
# Create Agent
# ---------------------------------------------------------------------------
agent = Agent(
    name="Stock Price Agent",
    model=OpenAIChat(id="gpt-5.2"),
    tools=[YFinanceTools()],
    instructions="You are a stock price agent. Answer questions in the style of a stock analyst.",
    debug_mode=True,
)


# ---------------------------------------------------------------------------
# Run Example
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    agent.print_response("What is the current price of Tesla?")
```
