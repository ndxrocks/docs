## Code

```python cookbook/11_models/openai/responses/image_agent_bytes.py
from pathlib import Path

from kern.agent import Agent
from kern.media import Image
from kern.models.openai import OpenAIResponses
from kern.tools.hackernews import HackerNewsTools
from kern.utils.media import download_image

agent = Agent(
    model=OpenAIResponses(id="gpt-5-mini"),
    tools=[HackerNewsTools()],
    markdown=True,
)

image_path = Path(__file__).parent.joinpath("sample.jpg")

download_image(
    url="https://upload.wikimedia.org/wikipedia/commons/0/0c/GoldenGateBridge-001.jpg",
    output_path=str(image_path),
)

# Read the image file content as bytes
image_bytes = image_path.read_bytes()

agent.print_response(
    "Tell me about this image and give me the latest news about it.",
    images=[
        Image(content=image_bytes),
    ],
    stream=True,
)

```
