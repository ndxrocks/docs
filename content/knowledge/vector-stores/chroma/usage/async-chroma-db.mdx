## Code

```python cookbook/08_knowledge/vector_db/chroma_db/async_chroma_db.py

import asyncio

from kern.agent import Agent
from kern.knowledge.knowledge import Knowledge
from kern.vectordb.chroma import ChromaDb

vector_db = ChromaDb(collection="recipes", path="tmp/chromadb", persistent_client=True)

knowledge = Knowledge(
    vector_db=vector_db,
)

agent = Agent(knowledge=knowledge)

if __name__ == "__main__":
    asyncio.run(
        knowledge.ainsert(url="https://kern.ndx.rocks/introduction/agents.md")
    )

    asyncio.run(
        agent.aprint_response("What is the purpose of an Kern Agent?", markdown=True)
    )
```
