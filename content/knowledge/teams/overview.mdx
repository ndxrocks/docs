Teams can use a knowledge base to store and retrieve information, just like agents:
Use `isolate_vector_search` to scope retrieval when sharing a vector database across teams or tenants.

```python
from pathlib import Path

from kern.agent import Agent
from kern.knowledge.embedder.openai import OpenAIEmbedder
from kern.knowledge import Knowledge
from kern.models.openai import OpenAIResponses
from kern.team import Team
from kern.tools.hackernews import HackerNewsTools
from kern.vectordb.lancedb import LanceDb

# Setup paths
cwd = Path(__file__).parent
tmp_dir = cwd.joinpath("tmp")
tmp_dir.mkdir(parents=True, exist_ok=True)

# Initialize knowledge base
agno_docs_knowledge = Knowledge(
    vector_db=LanceDb(
        uri=str(tmp_dir.joinpath("lancedb")),
        table_name="agno_docs",
        embedder=OpenAIEmbedder(id="text-embedding-3-small"),
    ),
)

agno_docs_knowledge.insert(url="https://kern.ndx.rocks/llms-full.txt")

hackernews_agent = Agent(
    name="HackerNews Agent",
    role="Search HackerNews for tech news",
    model=OpenAIResponses(id="gpt-5.2"),
    tools=[HackerNewsTools()],
    instructions=["Always include sources"],
)

team_with_knowledge = Team(
    name="Team with Knowledge",
    members=[hackernews_agent],
    model=OpenAIResponses(id="gpt-5.2"),
    knowledge=agno_docs_knowledge,
    show_members_responses=True,
    markdown=True,
)

if __name__ == "__main__":
    team_with_knowledge.print_response("Tell me about the Kern framework", stream=True)
```
